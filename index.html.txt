<!-- Firebase SDK Imports -->
<script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
    import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
    import { getFirestore, collection, onSnapshot, doc, setDoc, deleteDoc } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";
    import { setLogLevel } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

    setLogLevel('debug');

    // --- SECURITY CONSTANTS (Your Credentials) ---
    const ADMIN_USERNAME = 'hossam_hassan'; 
    const ADMIN_PASSWORD = 'hossammm';      

    // --- GLOBAL VARIABLES (Provided by Canvas Environment) ---
    // Note: The Firebase config variables below rely on the environment they are run in.
    // When deploying outside of this environment, you would typically replace the __* variables 
    // with your actual Firebase config object and anonymous sign-in logic.
    const appId = typeof __app_id !== 'undefined' ? __app_id : 'bus-finder-default-id';
    const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : { /* PLACE YOUR OWN CONFIG HERE IF DEPLOYING */ };
    const initialAuthToken = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;
    // --------------------------------------------------------

    if (!firebaseConfig || Object.keys(firebaseConfig).length === 0) {
        document.getElementById('loading-indicator').innerHTML = '<p class="text-red-500">Firebase configuration is missing. Cannot run application outside of the secure environment without manual config.</p>';
        throw new Error("Firebase configuration not found or empty.");
    }

    // Initialize Firebase
    const app = initializeApp(firebaseConfig);
    const db = getFirestore(app);
    const auth = getAuth(app);

    // DOM Elements
    const routeSelectEl = document.getElementById('route-select'); 
    const busNumberDisplayEl = document.getElementById('bus-number-display');
    const resultTextEl = document.getElementById('result-text');
    const loadingIndicatorEl = document.getElementById('loading-indicator');
    const mainContentEl = document.getElementById('main-content');
    const adminViewEl = document.getElementById('admin-view');
    const adminLoginBtn = document.getElementById('admin-login-btn');
    const adminLogoutBtn = document.getElementById('admin-logout-btn');
    const adminRouteListEl = document.getElementById('admin-route-list'); 
    const adminRouteIdEl = document.getElementById('admin-route-id');     
    const adminRouteNameEl = document.getElementById('admin-route-name'); 
    const adminBusNumberEl = document.getElementById('admin-bus-number');
    const adminSaveBtn = document.getElementById('admin-save-btn');
    const saveBtnTextEl = document.getElementById('save-btn-text');

    // Modal Elements
    const loginModalEl = document.getElementById('login-modal');
    const loginUsernameEl = document.getElementById('login-username');
    const loginPasswordEl = document.getElementById('login-password');
    const modalLoginBtn = document.getElementById('modal-login-btn');
    const modalCloseBtn = document.getElementById('modal-close-btn');
    const loginErrorEl = document.getElementById('login-error');

    // Global state
    let busAssignments = {};
    let isAuthReady = false;
    let isAdminLoggedIn = false; 

    // --- FIREBASE PATHS ---
    const BUS_ASSIGNMENTS_PATH = `artifacts/${appId}/public/data/bus_assignments`;

    /**
     * Helper function to log messages to the console for feedback.
     */
    function showMessage(title, message) {
        console.warn(`[${title}]: ${message}`);
    }

    /**
     * Authenticates the user and sets up the listener.
     */
    async function setupAuthAndDataListener() {
        try {
            if (initialAuthToken) {
                await signInWithCustomToken(auth, initialAuthToken);
            } else {
                await signInAnonymously(auth);
            }
        } catch(e) {
            console.error("Authentication failed:", e);
            loadingIndicatorEl.innerHTML = '<p class="text-red-500 font-bold">Authentication failed. Cannot load bus data.</p>';
            return;
        }

        onAuthStateChanged(auth, (user) => {
            if (user) {
                isAuthReady = true;
                listenForBusLanes();
            }
        });
    }

    /**
     * Sets up a real-time listener for bus assignment data.
     */
    function listenForBusLanes() {
        if (!isAuthReady) return;

        const q = collection(db, BUS_ASSIGNMENTS_PATH);

        onSnapshot(q, (snapshot) => {
            console.log("Received snapshot update. Updating UI.");
            busAssignments = {};

            // 1. Update Student View Dropdown
            routeSelectEl.innerHTML = '<option value="" disabled selected>-- Choose a Route --</option>'; 
            
            // 2. Update Admin View List
            adminRouteListEl.innerHTML = ''; 

            snapshot.forEach((doc) => {
                const data = doc.data();
                const routeId = doc.id; 
                busAssignments[routeId] = data;

                // Add to Student Dropdown
                const option = document.createElement('option');
                option.value = routeId;
                option.textContent = data.routeName || routeId; 
                routeSelectEl.appendChild(option); 

                // Add to Admin List (only if logged in)
                if (isAdminLoggedIn) {
                    const listItem = createAdminListItem(routeId, data);
                    adminRouteListEl.appendChild(listItem); 
                }
            });

            // Handle display logic
            loadingIndicatorEl.classList.add('hidden');
            mainContentEl.classList.remove('hidden');
            displayBusNumber(routeSelectEl.value); 

            // If the list is empty, prompt for data
            if (isAdminLoggedIn && snapshot.empty) {
                const emptyMsg = document.createElement('li');
                emptyMsg.className = "text-center text-gray-500 text-sm italic py-4";
                emptyMsg.textContent = "No routes found. Add your first route above!";
                adminRouteListEl.appendChild(emptyMsg); 
            }
        }, (error) => {
            console.error("Error fetching bus data:", error);
            loadingIndicatorEl.innerHTML = `<p class="text-red-500">Error loading data: ${error.message}</p>`;
        });
    }

    /**
     * Updates the UI to show the bus number for the selected route.
     * @param {string | null} routeId - The ID of the selected route.
     */
    function displayBusNumber(routeId) {
        if (!routeId || !busAssignments[routeId]) {
            resultTextEl.textContent = 'Select a route above to see your assigned bus number.';
            busNumberDisplayEl.textContent = '---';
            return;
        }

        const assignment = busAssignments[routeId];
        resultTextEl.textContent = `Your Bus Number for ${assignment.routeName} is:`; 
        busNumberDisplayEl.textContent = assignment.busNumber;
    }


    // --- ADMIN FUNCTIONS ---

    /**
     * Handles the login attempt.
     */
    function handleLogin() {
        loginErrorEl.classList.add('hidden');
        const username = loginUsernameEl.value;
        const password = loginPasswordEl.value;

        if (username === ADMIN_USERNAME && password === ADMIN_PASSWORD) {
            isAdminLoggedIn = true;
            loginModalEl.classList.add('hidden');
            adminViewEl.classList.remove('hidden');
            adminLoginBtn.classList.add('hidden'); // Hide login button
            
            // Clear inputs on successful login for cleanliness
            loginUsernameEl.value = '';
            loginPasswordEl.value = '';

            // Trigger a re-render of the list for the newly logged-in admin
            listenForBusLanes(); 
        } else {
            loginErrorEl.textContent = "Invalid username or password.";
            loginErrorEl.classList.remove('hidden');
            loginPasswordEl.value = ''; // Clear only password on failed attempt
        }
    }

    /**
     * Handles logging out.
     */
    function handleLogout() {
        isAdminLoggedIn = false;
        adminViewEl.classList.add('hidden');
        adminLoginBtn.classList.remove('hidden');
        // Clear admin form state
        adminRouteIdEl.value = '';
        adminRouteNameEl.value = '';
        adminBusNumberEl.value = '';
        adminRouteIdEl.readOnly = false;
        saveBtnTextEl.textContent = 'Add New Route';
        
        // Remove admin list content
        adminRouteListEl.innerHTML = '';
        showMessage("Logout", "You have been securely logged out of the Admin Panel.");
    }


    /**
     * Creates a list item for the Admin Panel.
     */
    function createAdminListItem(routeId, data) { 
        const li = document.createElement('li');
        li.className = "flex items-center justify-between p-3 bg-gray-50 rounded-lg shadow-sm";
        li.innerHTML = `
            <div class="flex-1 min-w-0">
                <p class="text-sm font-medium text-indigo-700">${data.routeName || routeId}</p>
                <p class="text-lg font-bold text-gray-900">${data.busNumber}</p>
            </div>
            <div class="space-x-2 flex-shrink-0 ml-4">
                <button data-route-id="${routeId}" class="admin-edit-btn text-sm px-3 py-1 bg-yellow-400 text-gray-800 rounded-md hover:bg-yellow-500 transition">Edit</button>
                <button data-route-id="${routeId}" class="admin-delete-btn text-sm px-3 py-1 bg-red-500 text-white rounded-md hover:bg-red-600 transition">Delete</button>
            </div>
        `;
        return li;
    }

    /**
     * Handles saving or updating a bus assignment.
     */
    async function handleSaveAssignment() {
        const routeId = adminRouteIdEl.value.trim();     
        const routeName = adminRouteNameEl.value.trim(); 
        const busNumber = adminBusNumberEl.value.trim();

        if (!routeId || !routeName || !busNumber) {
            showMessage("Input Error", "All fields (Route ID, Name, Bus Number) must be filled out.");
            return;
        }

        try {
            const docRef = doc(db, BUS_ASSIGNMENTS_PATH, routeId); 
            await setDoc(docRef, {
                routeName: routeName, 
                busNumber: busNumber,
            });
            showMessage("Success", `Assignment for ${routeName} saved successfully.`); 
            
            // Clear the form after success
            adminRouteIdEl.value = '';
            adminRouteNameEl.value = '';
            adminBusNumberEl.value = '';
            adminRouteIdEl.readOnly = false;
            saveBtnTextEl.textContent = 'Add New Route'; 

        } catch (error) {
            showMessage("Database Error", "Failed to save assignment. See console for details.");
            console.error("Error saving document:", error);
        }
    }
    
    /**
     * Handles deletion of a bus assignment.
     */
    async function handleDeleteAssignment(routeId) {
        try {
            const docRef = doc(db, BUS_ASSIGNMENTS_PATH, routeId);
            await deleteDoc(docRef);
            showMessage("Success", `Route ${routeId} deleted successfully.`);
        } catch (error) {
            showMessage("Database Error", "Failed to delete route. See console for details.");
            console.error("Error deleting document:", error);
        }
    }

    /**
     * Handles populating the form for editing.
     */
    function handleEditAssignment(routeId) {
        const assignment = busAssignments[routeId];
        if (!assignment) {
            showMessage("Error", "Assignment data not found.");
            return;
        }

        // Populate the form fields
        adminRouteIdEl.value = routeId;
        adminRouteNameEl.value = assignment.routeName; 
        adminBusNumberEl.value = assignment.busNumber;
        
        // Prevent editing the ID when updating
        adminRouteIdEl.readOnly = true;
        
        // Change button text to reflect update action
        saveBtnTextEl.textContent = 'Update Existing Route'; 

        // Scroll to the form
        adminViewEl.scrollIntoView({ behavior: 'smooth' });
    }

    // --- EVENT LISTENERS ---

    // Student View: Handle Route Selection
    routeSelectEl.addEventListener('change', (event) => { 
        displayBusNumber(event.target.value);
    });
    
    // Admin Login Button
    adminLoginBtn.addEventListener('click', () => {
        // Clear inputs every time the modal opens
        loginUsernameEl.value = '';
        loginPasswordEl.value = '';
        loginModalEl.classList.remove('hidden');
        loginUsernameEl.focus();
    });
    
    // Admin Logout Button
    adminLogoutBtn.addEventListener('click', handleLogout);

    // Modal Login Button
    modalLoginBtn.addEventListener('click', handleLogin);
    loginPasswordEl.addEventListener('keypress', (e) => {
        if (e.key === 'Enter') handleLogin();
    });

    // Modal Close Button (Clears both inputs)
    modalCloseBtn.addEventListener('click', () => {
        loginModalEl.classList.add('hidden');
        loginErrorEl.classList.add('hidden');
        loginUsernameEl.value = ''; 
        loginPasswordEl.value = ''; 
    });


    // Admin View: Save Button
    adminSaveBtn.addEventListener('click', handleSaveAssignment);

    // Admin View: Delegation for Edit/Delete buttons
    adminRouteListEl.addEventListener('click', (event) => { 
        const target = event.target;
        const routeId = target.dataset.routeId; 

        if (target.classList.contains('admin-delete-btn') && routeId) {
            handleDeleteAssignment(routeId);
        } else if (target.classList.contains('admin-edit-btn') && routeId) {
            handleEditAssignment(routeId);
        }
    });

    // Start the application
    setupAuthAndDataListener();

</script>
